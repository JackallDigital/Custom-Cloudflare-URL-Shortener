export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    const path = url.pathname;

    let token = getCookie(request, 'shortener_token');
    if (!token) {
      token = url.searchParams.get('token');
    }

    const departments = getDepartments(env);
    const allowedTokens = departments.map(d => d.token);

    const isLoggedIn = token && allowedTokens.includes(token);

    // === LOGIN / DASHBOARD ===
    if (path === '/login' || path === '/login/') {
      if (!isLoggedIn) {
        return htmlResponse(renderLoginPage(), 401);
      }

      const linkList = await getVisibleLinks(env, token, url.host);

      const departmentName = getDepartmentName(env, token);
      return htmlResponse(renderDashboardPage(linkList, {}, departmentName), 200, {
        'Cache-Control': 'no-store, no-cache, must-revalidate',
        'Pragma': 'no-cache',
        'Set-Cookie': buildSessionCookie(token)
      });
    }

    // === CREATE SHORT LINK ===
    if (path === '/shorten' && request.method === 'POST') {
      if (!isLoggedIn) {
        return Response.redirect(`${url.origin}/login`, 302);
      }

      const form = await request.formData();
      const longUrl = String(form.get('url') || '').trim();
      
      let slug = sanitizeSlug(form.get('slug'));

    if (!longUrl) {
      const linkList = await getVisibleLinks(env, token, url.host);

      return htmlResponse(
        renderDashboardPage(linkList, {
          longUrl,
          slug,
          slugError: 'Please enter a valid long URL.'
        }),
        400,
          {
            'Set-Cookie': buildSessionCookie(token)
          }
        );
    }

	  if (!slug) {
		  slug = await generateUniqueSlug(env, 7);
	  }
    else {
		  const existing = await env.KV.get(slug);

		  if (existing) {
        const linkList = await getVisibleLinks(env, token, url.host);

        return htmlResponse(
        renderDashboardPage(linkList, {
          longUrl,
          slug,
          slugError: 'This short code already exists. Please choose another one.'
        }),
        409,
        {
          'Set-Cookie': buildSessionCookie(token)
        }
        );
      }
	  }

    const createdAt = Date.now();
    const createdBy = getDepartmentName(env, token);

    const data = JSON.stringify({
      createdOn: new Date(createdAt).toISOString().slice(0, 10),
      createdBy,
      url: longUrl,
      createdAt
    });

	  await env.KV.put(slug, data);

	  const shortLink = `https://${url.host}/${slug}`;
	  const cacheBuster = Date.now();

	  return htmlResponse(
      renderCreatedPage(shortLink, cacheBuster),
      200,
      {
        'Set-Cookie': buildSessionCookie(token)
      }
    );
	}

    // === LOGOUT ===
    if (path === '/logout') {
      return htmlResponse(
        renderMessagePage(
          'You have been logged out',
          '<p class="message-text">Your session has ended successfully.</p><p><a class="link-button" href="/login">Go back to login</a></p>'
        ),
        200,
        {
          'Set-Cookie': clearSessionCookie(),
          'Cache-Control': 'no-store'
        }
      );
    }

    // === REDIRECT ===
    const slug = path.substring(1);
    if (slug && slug !== 'login' && slug !== 'logout' && slug !== 'shorten') {
      const stored = await env.KV.get(slug);
      if (stored) {
        try {
          const data = JSON.parse(stored);
          return Response.redirect(data.url, 301);
        } catch (e) {
          return Response.redirect(stored, 301);
        }
      }
    }

    return htmlResponse(
      renderMessagePage(
        'Shorties',
        '<p class="message-text">Welcome to Shorties.</p><p><a class="link-button" href="/login">Go to login</a></p>'
      )
    );
  }
};

function renderLinksList(linkList) {
  if (!linkList.length) {
    return `<p class="empty-state">No links found yet.</p>`;
  }

  const items = linkList.map((item) => {
    const hasFullLink = item.fullUrl.length > 35;
    const id = item.createdAt;
    const shortUrlSafe = escapeHtml(item.shortUrl);
    const displayLongSafe = escapeHtml(item.displayLong);
    const fullUrlJs = escapeJsString(item.fullUrl);

    return `
      <li class="link-item">
        <div class="link-row">
          <div class="link-main">
            <div class="short-url">${shortUrlSafe}</div>
            <div class="long-url-row">
              <span id="long-${id}" class="long-url-text">${displayLongSafe}</span>
              ${
                hasFullLink
                  ? `<a
                      href="#"
                      id="toggle-${id}"
                      class="text-link"
                      data-expanded="false"
                      onclick="toggleFull('${fullUrlJs}', ${id}); return false;"
                    >Show full</a>`
                  : ''
              }
            </div>
          </div>

          <div class="link-actions">
            <button
              type="button"
              class="button button-secondary button-small"
              onclick="copyToClipboard('${escapeJsString(item.shortUrl)}')"
            >
              Copy
            </button>
          </div>
        </div>
      </li>
    `;
  }).join('');

  return `<ul class="links-list">${items}</ul>`;
}

function htmlResponse(html, status = 200, extraHeaders = {}) {
  return new Response(html, {
    status,
    headers: {
      'Content-Type': 'text/html; charset=UTF-8',
      ...extraHeaders
    }
  });
}

function buildSessionCookie(token) {
  return `shortener_token=${token}; Path=/; Max-Age=1800; HttpOnly; SameSite=Strict`;
}

function clearSessionCookie() {
  return `shortener_token=; Path=/; Max-Age=0; Expires=Thu, 01 Jan 1970 00:00:00 GMT`;
}

function getCookie(request, name) {
  const cookieHeader = request.headers.get('Cookie');
  if (!cookieHeader) return null;

  const cookies = cookieHeader.split(';').map((c) => c.trim());
  for (const cookie of cookies) {
    if (cookie.startsWith(name + '=')) {
      return cookie.substring(name.length + 1);
    }
  }

  return null;
}

function escapeHtml(value) {
  return String(value)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}

function escapeJsString(value) {
  return String(value)
    .replace(/\\/g, '\\\\')
    .replace(/'/g, "\\'")
    .replace(/\r/g, '\\r')
    .replace(/\n/g, '\\n')
    .replace(/<\/script>/gi, '<\\/script>');
}

async function getVisibleLinks(env, token, host) {
  const { keys } = await env.KV.list({ limit: 100 });
  const linkList = [];
  const isAdmin = token === env.ADMIN_TOKEN;
  const currentDepartment = getDepartmentName(env, token);

  for (const key of keys) {
    const stored = await env.KV.get(key.name);
    if (!stored) continue;

    let longUrl = '';
    let createdAt = 0;
    let createdBy = '';

    try {
      const data = JSON.parse(stored);
      longUrl = data.url || stored;
      createdAt = data.createdAt || Date.now();
      createdBy = data.createdBy || '';
    } catch (e) {
      longUrl = stored || '';
      createdAt = Date.now();
    }

    if (!isAdmin && createdBy !== currentDepartment) continue;

    const shortUrl = `https://${host}/${key.name}`;
    const displayLong =
      longUrl.length > 35 ? longUrl.substring(0, 35) + '...' : longUrl;

    linkList.push({
      shortUrl,
      fullUrl: longUrl,
      displayLong,
      createdAt
    });
  }

  linkList.sort((a, b) => b.createdAt - a.createdAt);
  return linkList;
}

async function generateUniqueSlug(env, length = 7) {
  const maxAttempts = 20;

  for (let i = 0; i < maxAttempts; i++) {
    const slug = generateRandomSlug(length);
    const existing = await env.KV.get(slug);

    if (!existing) {
      return slug;
    }
  }

  throw new Error('Could not generate a unique slug.');
}

function generateRandomSlug(length = 7) {
  const alphabet = 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
  const bytes = new Uint8Array(length);
  crypto.getRandomValues(bytes);

  let result = '';
  for (let i = 0; i < length; i++) {
    result += alphabet[bytes[i] % alphabet.length];
  }

  return result;
}

function getDepartments(env) {
  return [
    { token: env.ADMIN_TOKEN, name: 'ADMIN' },
    { token: env.MARKETING_TOKEN, name: 'MARKETING' },
    { token: env.IT_TOKEN, name: 'IT' },
    { token: env.SALES_TOKEN, name: 'SALES' },
    { token: env.PRODUCTION_TOKEN, name: 'PRODUCTION' },
    { token: env.LOGISTICS_TOKEN, name: 'LOGISTICS' }
  ].filter(d => d.token);
}

function getDepartmentName(env, token) {
  if (token === env.ADMIN_TOKEN) return 'Admin';
  if (token === env.MARKETING_TOKEN) return 'Marketing';
  if (token === env.IT_TOKEN) return 'IT';
  if (token === env.SALES_TOKEN) return 'Sales';
  if (token === env.PRODUCTION_TOKEN) return 'Production';
  if (token === env.LOGISTICS_TOKEN) return 'Logistics';
  return 'Unknown';
}

function sanitizeSlug(input) {
  return String(input || '')
    .trim()
    //.toLowerCase()
    .replace(/\s+/g, '-')         // spaces -> single dash
    .replace(/[^a-zA-Z0-9-]/g, '') // remove everything except letters, numbers, dash
    .replace(/-+/g, '-')          // collapse repeated dashes
    .replace(/^-|-$/g, '');       // trim leading/trailing dashes
}
